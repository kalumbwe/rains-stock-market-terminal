   150→const round2 = (x: number): number => Math.round(x * 100) / 100
   151→const rnd = (a: number, b: number): number => a + Math.random() * (b - a)
   152→const pick = <T>(arr: readonly T[]): T => arr[Math.floor(Math.random() * arr.length)]
   153→const chance = (p: number): boolean => Math.random() < p

   155→/** Standard normal sample (Box–Muller). */
   156→function gauss(): number {
   157→  let u = 0
   158→  let v = 0
   159→  while (u === 0) u = Math.random()
   160→  while (v === 0) v = Math.random()
   161→  return Math.sqrt(-2 * Math.log(u)) * Math.cos(2 * Math.PI * v)
   162→}

   186→  // Liquidity proxy in [0,1]: log market-cap rank. Bigger cap → tighter spread.
   187→  const liquidity = maxCap > minCap ? Math.log(caps[i] / minCap) / Math.log(maxCap / minCap) : 1
   188→  const sigmaAnnual = 0.15 + 0.13 * Math.abs(cfg.beta)
   189→  const st: StockState = {
   190→    cfg,
   191→    price: cfg.price,
   192→    prevClose: cfg.prevClose,
   193→    dayOpen: round2(cfg.prevClose * (1 + (chance(0.5) ? 1 : -1) * rnd(0, 0.003))),
   194→    dayHigh: cfg.price,
   195→    dayLow: cfg.price,
   196→    volume: 0,
   197→    valueTraded: 0,
   198→    bid: cfg.price,
   199→    ask: cfg.price,
   200→    anchor: cfg.price,
   201→    spreadPct: 0.0025 - 0.0015 * liquidity, // 0.1%–0.25%
   202→    sigmaTick: sigmaAnnual * SIGMA_STEP_SCALE,
   203→    volScale: 0.25 + 1.75 * liquidity,

   211→  st.ask = round2(st.price * (1 + st.spreadPct / 2))

   233→function lusakaParts(ts: number): LusakaParts {
   234→  const d = new Date(ts + TZ_OFFSET_MIN * 60_000)
   235→  return {
   236→    y: d.getUTCFullYear(),
   237→    mo: d.getUTCMonth(),
   238→    d: d.getUTCDate(),
   239→    h: d.getUTCHours(),
   240→    mi: d.getUTCMinutes(),
   241→    s: d.getUTCSeconds(),
   242→    dow: d.getUTCDay(),
   243→  }
   244→}

   246→const pad2 = (n: number): string => String(n).padStart(2, '0')

   248→function lusakaClock(ts: number): string {
   249→  const p = lusakaParts(ts)
   250→  return `${pad2(p.h)}:${pad2(p.mi)}:${pad2(p.s)}`
   251→}

   253→/** OPEN on weekdays 09:30–15:30 Lusaka, else CLOSED. */
   254→function isInSession(ts: number): boolean {
   255→  const p = lusakaParts(ts)
   256→  if (p.dow === 0 || p.dow === 6) return false
   257→  const mins = p.h * 60 + p.mi
   258→  const open = meta.session.openHourLocal * 60 + meta.session.openMinLocal
   259→  const close = meta.session.closeHourLocal * 60 + meta.session.closeMinLocal
   260→  return mins >= open && mins < close
   261→}

   263→// ---------------------------------------------------------------------------
   264→// Session candle seeding — GBM walk per stock, bridged to end at cfg.price
   265→// ---------------------------------------------------------------------------
   266→function seedSession(): number {
   267→  const now = Date.now()
   268→  let startMs: number
   269→  if (isInSession(now)) {
   270→    const p = lusakaParts(now)
   271→    const openMinAbs = meta.session.openHourLocal * 60 + meta.session.openMinLocal
   272→    startMs = Date.UTC(p.y, p.mo, p.d) + (openMinAbs - TZ_OFFSET_MIN) * 60_000
   273→  } else {
   274→    // Outside hours / weekend: virtual session so charts are never empty.
   275→    startMs = now - VIRTUAL_SESSION_HOURS * 3_600_000
   276→  }
   277→  const startBucket = Math.floor(startMs / 60_000) * 60_000
   278→  const nowBucket = Math.floor(now / 60_000) * 60_000
   279→  const nMinutes = Math.max(0, Math.round((nowBucket - startBucket) / 60_000))
   280→
   281→  // Per-stock minute price paths (index 0..nMinutes). A Brownian-bridge style
   282→  // correction on the log path guarantees the walk ends exactly at cfg.price.
   283→  const paths: number[][] = stocks.map((s) => {
   284→    const path = [s.dayOpen]
   285→    if (nMinutes > 0) {
   286→      const logs = [Math.log(s.dayOpen)]
   287→      for (let m = 1; m <= nMinutes; m++) logs.push(logs[m - 1] + s.sigmaTick * gauss())
   288→      const end = Math.log(s.cfg.price)
   289→      for (let m = 1; m <= nMinutes; m++) {
   290→        path.push(Math.exp(logs[m] + ((end - logs[nMinutes]) * m) / nMinutes))
   291→      }
   292→    }
   293→    return path
   294→  })
   295→
   296→  // Build 1m candles + running volume/valueTraded, and day high/low.
   297→  stocks.forEach((s, i) => {
   298→    const p = paths[i]
   299→    for (let m = 0; m <= nMinutes; m++) {
   300→      const t = startBucket + m * 60_000
   301→      const o = round2(m === 0 ? p[0] : p[m - 1])
   302→      const c = round2(p[m])
   303→      const h = round2(Math.max(o, c) * (1 + rnd(0, 0.0015)))
   304→      const l = round2(Math.min(o, c) * (1 - rnd(0, 0.0015)))
   305→      const v = Math.max(1, Math.floor(rnd(50, 15_000) * s.volScale))
   306→      s.candles.push({ t, o, h, l, c, v })
   307→      s.volume += v
   308→      s.valueTraded += v * c
   309→    }
   310→    s.dayHigh = round2(Math.max(...s.candles.map((c) => c.h), s.price))
   311→    s.dayLow = round2(Math.min(...s.candles.map((c) => c.l), s.price))
   312→  })
   313→
   314→  // Seed the index history from the same minute paths.
   315→  for (let m = 0; m <= nMinutes; m++) {
   316→    let rel = 0
   317→    for (let i = 0; i < stocks.length; i++) rel += stocks[i].weight * (paths[i][m] / stocks[i].anchor)
   318→    indexHistory.push({ t: startBucket + m * 60_000, v: round2(indexBase * rel) })
   319→  }
   320→  return nMinutes
   321→}

   323→// ---------------------------------------------------------------------------
   324→// LASI index

   326→function currentIndexValue(): number {
   327→  let rel = 0
   328→  for (const s of stocks) rel += s.weight * (s.price / s.anchor)
   329→  return round2(indexBase * rel)
   330→}

   336→)

   338→// ---------------------------------------------------------------------------
   339→// Order books — regenerated every few seconds, stable in between
   340→// ---------------------------------------------------------------------------
   341→function regenBooks(): void {
   342→  for (const s of stocks) {
   343→    const bids: BookLevel[] = []
   344→    const asks: BookLevel[] = []
   345→    for (let i = 0; i < 5; i++) {
   346→      // Level offsets: level 1 ≈ 0.05%, deepening to ≤ 0.30% at level 5.
   347→      const off = Math.min(0.003, (i + 1) * 0.0005 + rnd(0, 0.0004))
   348→      const bidRef = bids.length ? bids[bids.length - 1].price : s.price
   349→      let bp = round2(s.price * (1 - off))
   350→      if (bp >= bidRef) bp = round2(bidRef - 0.01) // keep ladder strict after rounding
   351→      bids.push({ price: Math.max(0.01, bp), size: 100 + Math.floor(Math.random() * 49_901) })
   352→
   353→      const askRef = asks.length ? asks[asks.length - 1].price : s.price
   354→      let ap = round2(s.price * (1 + off))
   355→      if (ap <= askRef) ap = round2(askRef + 0.01)
   356→      asks.push({ price: ap, size: 100 + Math.floor(Math.random() * 49_901) })
   357→    }
   358→    s.book = { bids, asks }
   359→  }
   360→}

   637→/** Renders a random Zambia-themed headline. */
   638→function generateNews(): NewsItem {
   639→  const finalize = (raw: RawNews, symbols: string[]): NewsItem => {
   640→    // Kwacha headlines nudge the FX rate immediately.
   641→    if (raw.usdDelta !== undefined) {
   642→      usdRate = round2(Math.min(60, Math.max(5, usdRate * (1 + raw.usdDelta))))
   643→    }
   644→    return {
   645→      id: `n-${Date.now()}-${Math.random().toString(36).slice(2, 6)}`,
   646→      headline: raw.headline,
   647→      body: raw.body,
   648→      source: 'LuSE Wire',
   649→      symbols,
   650→      sentiment: raw.sentiment,
   651→      impact: raw.impact,
   652→      publishedAt: new Date().toISOString(),
   653→    }
   654→  }
   655→
   656→  if (chance(0.55)) {
   657→    // Company news: always tagged with the related symbol (sometimes a peer).
   658→    const s = pick(stocks)
   659→    const symbols = chance(0.15) ? [s.cfg.symbol, pick(stocks).cfg.symbol] : [s.cfg.symbol]
   660→    return finalize(pick(companyTemplates(s))(), symbols)
   661→  }
   662→  // Macro news: templates either tag related symbols or stay [] for macro.
   663→  const raw = pick(macroTemplates())()
   664→  return finalize(raw, raw.symbols ?? [])
   665→}

   667→/** Queue an immediate price jump for the affected stocks (applied next tick). */
   668→function applyNewsImpact(item: NewsItem): void {
   669→  const sign =
   670→    item.sentiment === 'POSITIVE' ? 1 : item.sentiment === 'NEGATIVE' ? -1 : chance(0.5) ? 1 : -1
   671→  const [lo, hi] = IMPACT_RANGE[item.impact]
   672→  const base = rnd(lo, hi)
   673→  const marketWide = item.symbols.length === 0
   674→  // Macro news (no symbols) moves the whole tape at roughly half strength.
   675→  const targets = marketWide ? stocks.map((s) => s.cfg.symbol) : item.symbols
   676→  const scale = marketWide ? 0.5 : 1
   677→  for (const symbol of targets) {
   678→    const pct = sign * base * scale * rnd(0.6, 1.4)
   679→    pendingImpacts.set(symbol, (pendingImpacts.get(symbol) ?? 0) + pct)
   680→  }
   681→}

   684→  const item = generateNews()
   685→  applyNewsImpact(item)
   686→  newsLog.unshift(item)
   687→  if (newsLog.length > NEWS_CAP) newsLog.pop()
   688→  io.emit('news', item)
   689→  console.log(`[news] (${item.sentiment}/${item.impact}) ${item.headline}`)
   690→}

   703→function tickVolume(s: StockState): number {
   704→  return Math.max(1, Math.floor(rnd(50, 15_000) * s.volScale))
   705→}
   706→
   707→function tick(): void {
   708→  lastTickAt = Date.now()
   709→  const now = Date.now()
   710→  const bucket = Math.floor(now / 60_000) * 60_000
   711→
   712→  for (const s of stocks) {
   713→    // 1) Price: GBM step + weak mean reversion + rare jump + queued news impact.
   714→    const pending = pendingImpacts.get(s.cfg.symbol)
   715→    if (pending !== undefined) pendingImpacts.delete(s.cfg.symbol)
   716→    const reversion = MEAN_REVERSION_K * Math.log(s.anchor / s.price)
   717→    const jump = chance(JUMP_PROB) ? (chance(0.5) ? -1 : 1) * rnd(0.004, 0.012) : 0
   718→    const ret = reversion + s.sigmaTick * gauss() + jump + (pending ?? 0)
   719→
   720→    const prev = s.price
   721→    s.price = round2(Math.max(0.01, s.price * Math.exp(ret)))
   722→    if (s.price > s.dayHigh) s.dayHigh = s.price
   723→    if (s.price < s.dayLow) s.dayLow = s.price
   724→    s.dir = s.price > prev ? 1 : s.price < prev ? -1 : 0
   725→
   726→    // 2) Volume & value traded.
   727→    const vol = tickVolume(s)
   728→    s.volume += vol
   729→    s.valueTraded += vol * s.price
   730→
   731→    // 3) Re-quote bid/ask around the new price with a little jitter.
   732→    const half = s.price * (s.spreadPct / 2) * rnd(0.85, 1.15)
   733→    s.bid = round2(s.price - half)
   734→    if (s.bid >= s.price) s.bid = round2(s.price - 0.01)
   735→    s.bid = Math.max(0.01, s.bid)
   736→    s.ask = round2(s.price + half)
   737→    if (s.ask <= s.price) s.ask = round2(s.price + 0.01)
   738→    s.lastUpdate = new Date(now).toISOString()
   739→
   740→    // 4) Live 1m candle bucket.
   741→    const last = s.candles[s.candles.length - 1]
   742→    if (last && last.t === bucket) {
   743→      last.c = s.price
   744→      if (s.price > last.h) last.h = s.price
   745→      if (s.price < last.l) last.l = s.price
   746→      last.v += vol
   747→    } else {
   748→      s.candles.push({ t: bucket, o: s.price, h: s.price, l: s.price, c: s.price, v: vol })
   749→      if (s.candles.length > CANDLE_CAP) s.candles.shift()
   750→    }
   751→  }
   752→
   753→  // 5) USD/ZMW random walk: ±0.05% steps with a gentle pull to the meta anchor.
   754→  usdRate = round2(usdRate * (1 + rnd(-0.0005, 0.0005)) + (meta.usdRate - usdRate) * 0.02)
   755→
   756→  // 6) Index point per tick (cap history length).
   757→  const idxValue = currentIndexValue()
   758→  indexHistory.push({ t: now, v: idxValue })
   759→  if (indexHistory.length > INDEX_HISTORY_CAP) {
   760→    indexHistory.splice(0, indexHistory.length - INDEX_HISTORY_CAP)
   761→  }
   762→
   763→  io.emit('tick', buildTick(now, idxValue))
   764→}

   769→function indexBlock(idxValue: number, withHistory: boolean) {
   770→  const change = round2(idxValue - indexPrevCloseValue)
   771→  return {
   772→    code: meta.indexCode,
   773→    value: idxValue,
   774→    prevClose: indexPrevCloseValue,
   775→    change,
   776→    changePct: round2((change / indexPrevCloseValue) * 100),
   777→    ...(withHistory ? { history: indexHistory } : {}),
   778→  }
   779→}

   847→const CORS_HEADERS: Record<string, string> = {
   848→  'Access-Control-Allow-Origin': '*',
   849→  'Access-Control-Allow-Methods': 'GET, OPTIONS',
   850→  'Access-Control-Allow-Headers': 'Content-Type',

   853→function sendJson(res: ServerResponse, status: number, payload: unknown): void {
   854→  const body = JSON.stringify(payload)
   855→  res.writeHead(status, {
   856→    ...CORS_HEADERS,
   857→    'Content-Type': 'application/json; charset=utf-8',
   858→    'Cache-Control': 'no-store',
   859→    'Content-Length': Buffer.byteLength(body),
   860→  })
   861→  res.end(body)
   862→}

   864→function handleRequest(req: IncomingMessage, res: ServerResponse): void {
   865→  try {
   866→    const url = new URL(req.url ?? '/', `http://${req.headers.host ?? 'localhost'}`)
   867→    const path = url.pathname
   868→
   869→    if (req.method === 'OPTIONS') {
   870→      res.writeHead(204, CORS_HEADERS)
   871→      res.end()
   872→      return
   873→    }
   874→    if (req.method !== 'GET') {
   875→      sendJson(res, 405, { error: 'method-not-allowed' })
   876→      return
   877→    }
   878→
   879→    if (path === '/health') {
   880→      const tickAgeMs = lastTickAt > 0 ? Date.now() - lastTickAt : null
   881→      sendJson(res, 200, {
   882→        ok: true,
   883→        serverTime: new Date().toISOString(),
   884→        uptimeSec: Math.floor((Date.now() - bootMs) / 1000),
   885→        tickAgeMs,
   886→        ticking: tickAgeMs === null ? false : tickAgeMs < 10_000,
   887→      })
   888→      return
   889→    }
   890→
   891→    if (path === '/api/stocks') {
   892→      sendJson(res, 200, getSnapshot())
   893→      return
   894→    }
   895→
   896→    if (path === '/api/news') {
   897→      const raw = url.searchParams.get('limit')
   898→      let limit = 40
   899→      if (raw !== null) {
   900→        limit = Number(raw)
   901→        if (!Number.isInteger(limit) || limit < 1) {
   902→          sendJson(res, 400, { error: 'invalid-limit' })
   903→          return
   904→        }
   905→      }
   906→      limit = Math.min(limit, NEWS_CAP)
   907→      sendJson(res, 200, { news: newsLog.slice(0, limit) })
   908→      return
   909→    }
   910→
   911→    const m = path.match(/^\/api\/stocks\/([^/]+)\/(candles|orderbook)$/)
   912→    if (m) {
   913→      const symbol = decodeURIComponent(m[1]).toUpperCase()
   914→      const s = stockBySymbol.get(symbol)
   915→      if (!s) {
   916→        sendJson(res, 404, { error: 'stock-not-found' })
   917→        return
   918→      }
   919→      if (m[2] === 'orderbook') {
   920→        sendJson(res, 200, { symbol: s.cfg.symbol, bids: s.book.bids, asks: s.book.asks })
   921→        return
   922→      }
   923→      const interval = url.searchParams.get('interval') ?? '1m'
   924→      if (interval !== '1m') {
   925→        sendJson(res, 400, { error: 'invalid-interval', message: 'only interval=1m is supported' })
   926→        return
   927→      }
   928→      sendJson(res, 200, { symbol: s.cfg.symbol, interval: '1m', candles: s.candles })
   929→      return
   930→    }
   931→
   932→    sendJson(res, 404, { error: 'not-found' })
   933→  } catch (err) {
   934→    console.error('[http] error handling request:', err)
   935→    sendJson(res, 500, { error: 'internal-error' })
   936→  }
   937→}
